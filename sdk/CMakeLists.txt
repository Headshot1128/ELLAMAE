cmake_minimum_required(VERSION 3.20)
project(secureauth VERSION 1.0.0 LANGUAGES CXX)
find_package(CURL 7.85 REQUIRED)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
add_library(secureauth src/client.cpp)
target_include_directories(secureauth PUBLIC include)
target_compile_features(secureauth PUBLIC cxx_std_17)
target_link_libraries(secureauth PUBLIC CURL::libcurl nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(secureauth PRIVATE /W4)
else()
  target_compile_options(secureauth PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(auth_example examples/main.cpp)
target_link_libraries(auth_example PRIVATE secureauth)
if(WIN32)
  target_link_libraries(auth_example PRIVATE advapi32)
endif()
enable_testing()
add_executable(sdk_tests tests/client_test.cpp)
target_link_libraries(sdk_tests PRIVATE secureauth)
add_test(NAME sdk_fail_closed COMMAND sdk_tests)
add_executable(sdk_e2e tests/e2e.cpp)
target_link_libraries(sdk_e2e PRIVATE secureauth)
add_executable(sdk_live_probe tests/live_probe.cpp)
target_link_libraries(sdk_live_probe PRIVATE secureauth)
